\newglossaryentry{stem}{name={stem},description={An isolated component of a mixed recording, such as vocals, drums or bass}}
\newglossaryentry{residual}{name={residual},description={The part of the original mix not captured by any produced stem: mix minus the sum of all stems}}
\newglossaryentry{preset}{name={preset},description={A named separation recipe (a graph of model runs) such as Fast, Balanced or Max quality}}
\newglossaryentry{backingtrack}{name={backing track},description={A rendered mix of the song without the stems the user removes, used to play or sing along}}
